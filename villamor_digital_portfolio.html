finmansys_core.sql
NORMALIZATION: 3NF VERIFIED ENGINE: InnoDB / UTF8MB4
users ENTITY
  • PK user_id INT UNSIGNED
  • full_name VARCHAR(100)
  • UQ email VARCHAR(120)
  • created_at TIMESTAMP
accounts 1:N USERS
  • PK account_id INT UNSIGNED
  • FK user_id → users.id
  • account_type ENUM
  • balance DECIMAL(12,2)
transactions LEDGER / FK INTEGRITY
  • PK tx_id BIGINT UNSIGNED
  • FK account_id → accounts.id
  • entry_type ENUM('CR','DR')
  • amount / recorded_at DECIMAL(12,2)
QUERY: Account Ledger Summary INDEXED JOIN 
SELECT
  u.full_name,
  a.account_type,
  COUNT(t.tx_id) AS total_entries,
  COALESCE(SUM(
    CASE WHEN t.entry_type = 'CR' THEN t.amount ELSE -t.amount END
  ), 0.00) AS net_flow
FROM users u
INNER JOIN accounts a ON u.user_id = a.user_id
LEFT JOIN transactions t ON a.account_id = t.account_id
GROUP BY u.user_id, a.account_id
ORDER BY net_flow DESC;
